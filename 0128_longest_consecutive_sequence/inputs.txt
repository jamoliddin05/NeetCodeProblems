# Test cases for Longest Consecutive Sequence — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [5] * 100.
# `random` is available in expressions and is seeded, so random data is the same on every run.
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [2, 20, 4, 10, 3, 4, 5]
expected = 4

# Example 2
nums = [0, 3, 2, 5, 4, 6, 1, 1]
expected = 7

# empty array
nums = []
expected = 0

# single element
nums = [7]
expected = 1

# all values equal: duplicates don't extend a run
nums = [5, 5, 5, 5]
expected = 1

# duplicates inside a run
nums = [1, 2, 2, 3, 3, 3, 4]
expected = 4

# run crosses zero, negatives, unsorted
nums = [-2, 0, -1, 1, -3]
expected = 5

# gaps of 2: nothing is consecutive
nums = [9, 1, 7, 3, 5]
expected = 1

# two runs, the longer one comes second and is reverse-sorted
nums = [1, 4, 2, 3, 200, 105, 104, 103, 102, 101, 100]
expected = 6

# constraint extremes: an array indexed by value won't fit
nums = [1000000000, -1000000000, 999999999, -999999999, 0]
expected = 2

# stress: one shuffled run of 10^5 distinct values
nums = random.sample(range(-50000, 50000), 100000)
expected = 100000

# stress: the run's first value repeated 50,000 times
nums = [0] * 50000 + list(range(50000))
expected = 50000
