# Test cases for Longest Common Subsequence — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like "a" * 1000.
# `random` in an expression is seeded from the expression's text, so the data is the same on every run.
# Leave `expected` out to just see what your solution returns.

# Example 1
text1 = "abcde"
text2 = "ace"
expected = 3

# Example 2
text1 = "abc"
text2 = "abc"
expected = 3

# Example 3
text1 = "abc"
text2 = "def"
expected = 0

# single equal letters
text1 = "a"
text2 = "a"
expected = 1

# single different letters
text1 = "a"
text2 = "b"
expected = 0

# same letters in reverse order
text1 = "abc"
text2 = "cba"
expected = 1

# repeated letters
text1 = "aaaa"
text2 = "aa"
expected = 2

# pairing the first letter as early as possible loses
text1 = "dabc"
text2 = "abcd"
expected = 3

# 1000 letters each, nothing in common: plain recursion is too slow
text1 = "a" * 1000
text2 = "b" * 1000
expected = 0

# 1000 random letters each
text1 = "".join(random.choices("abcde", k=1000))
text2 = "".join(random.choices("edcba", k=1000))
expected = 606
