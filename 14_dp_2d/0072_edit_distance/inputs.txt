# Test cases for Edit Distance — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like "a" * 500.
# `random` in an expression is seeded from the expression's text, so the data is the same on every run.
# Leave `expected` out to just see what your solution returns.

# Example 1
word1 = "horse"
word2 = "ros"
expected = 3

# Example 2
word1 = "intention"
word2 = "execution"
expected = 5

# both words empty
word1 = ""
word2 = ""
expected = 0

# first word empty: insert everything
word1 = ""
word2 = "abc"
expected = 3

# second word empty: delete everything
word1 = "abc"
word2 = ""
expected = 3

# equal words
word1 = "same"
word2 = "same"
expected = 0

# no letters in common: replace each one
word1 = "abc"
word2 = "xyz"
expected = 3

# shifted by one letter: one delete and one insert
word1 = "abcdef"
word2 = "bcdefg"
expected = 2

# different lengths: replaces and an insert
word1 = "kitten"
word2 = "sitting"
expected = 3

# 500 a's to 500 b's: plain recursion is too slow
word1 = "a" * 500
word2 = "b" * 500
expected = 500

# 500 random letters each
word1 = "".join(random.choices("abc", k=500))
word2 = "".join(random.choices("cba", k=500))
expected = 229
