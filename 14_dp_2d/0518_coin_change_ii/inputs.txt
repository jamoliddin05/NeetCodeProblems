# Test cases for Coin Change II — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(2, 601, 2)).
# Leave `expected` out to just see what your solution returns.

# Example 1
amount = 5
coins = [1, 2, 5]
expected = 4

# Example 2
amount = 3
coins = [2]
expected = 0

# Example 3
amount = 10
coins = [10]
expected = 1

# amount 0: one way, use no coins
amount = 0
coins = [7]
expected = 1

# a different order of the same coins is not a new way
amount = 3
coins = [1, 2]
expected = 2

# every coin is larger than the amount
amount = 3
coins = [5, 10]
expected = 0

# coins not sorted
amount = 10
coins = [5, 1, 2]
expected = 10

# many combinations: plain recursion is too slow
amount = 500
coins = [3, 5, 7, 8, 9, 10, 11]
expected = 35502874

# largest amount with coins 1, 2 and 5: plain recursion is too slow
amount = 5000
coins = [1, 2, 5]
expected = 1252001

# 300 even coins, odd amount: no way, plain recursion is too slow
amount = 4999
coins = list(range(2, 601, 2))
expected = 0

# 300 coins, largest amount: only the coin 5000 fits
amount = 5000
coins = list(range(4701, 5001))
expected = 1
