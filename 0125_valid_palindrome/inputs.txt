# Test cases for Valid Palindrome — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like "ab" * 500.
# Leave `expected` out to just see what your solution returns.

# Example 1
s = "Was it a car or a cat I saw?"
expected = true

# Example 2
s = "tab a cat"
expected = false

# single character
s = "a"
expected = true

# only non-alphanumeric: nothing left counts as a palindrome
s = " .,!? "
expected = true

# digits count, and differ
s = "0P"
expected = false

# digits that mirror around letters
s = "1a2b2a1"
expected = true

# case is ignored
s = "AbBa"
expected = true

# almost a palindrome: one middle pair differs
s = "race a car"
expected = false

# punctuation clustered at one end only
s = "!!!ab,a"
expected = true

# two letters, different
s = "ab"
expected = false

# max length: 1000 chars, palindrome with mixed case and symbols
s = ("Ab1-" * 125) + ("-1bA" * 125)
expected = true

# max length: mismatch only at the exact center
s = "x" * 499 + "yz" + "x" * 499
expected = false
