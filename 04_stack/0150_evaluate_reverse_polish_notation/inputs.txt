# Test cases for Evaluate Reverse Polish Notation — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like ["1"] * 5000 + ["+"] * 4999.
# Leave `expected` out to just see what your solution returns.

# Example 1
tokens = ["2", "1", "+", "3", "*"]
expected = 9

# Example 2
tokens = ["4", "13", "5", "/", "+"]
expected = 6

# Example 3
tokens = ["10", "6", "9", "3", "+", "-11", "*", "/", "*", "17", "+", "5", "+"]
expected = 22

# single number
tokens = ["42"]
expected = 42

# single negative number is not an operator
tokens = ["-7"]
expected = -7

# subtracting a negative number
tokens = ["5", "-3", "-"]
expected = 8

# operand order for minus
tokens = ["3", "10", "-"]
expected = -7

# operand order for division
tokens = ["2", "10", "/"]
expected = 0

# negative result of division truncates toward zero
tokens = ["7", "-2", "/"]
expected = -3

# negative divided by positive truncates toward zero
tokens = ["-7", "2", "/"]
expected = -3

# stress: 5000 numbers, then 4999 plus signs
tokens = ["1"] * 5000 + ["+"] * 4999
expected = 5000

# stress: 9999 tokens, 200 minus -200, repeated 4999 times
tokens = ["200"] + ["-200", "-"] * 4999
expected = 1000000
