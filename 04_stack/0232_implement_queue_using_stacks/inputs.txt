# Test cases for Implement Queue using Stacks — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each line is one call on the same object; the first line creates it.
# `-> value` after a call is the value it must return; calls without `->` aren't checked.
# Values are JSON (true, false, null, "text") or Python expressions.

# Example 1
MyQueue()
push(1)
push(2)
peek() -> 1
pop() -> 1
empty() -> false

# new queue is empty
MyQueue()
empty() -> true

# single element in and out
MyQueue()
push(5)
empty() -> false
peek() -> 5
pop() -> 5
empty() -> true

# first in, first out over several elements
MyQueue()
push(1)
push(2)
push(3)
pop() -> 1
pop() -> 2
pop() -> 3
empty() -> true

# peek doesn't remove anything
MyQueue()
push(4)
push(8)
peek() -> 4
peek() -> 4
pop() -> 4
peek() -> 8

# pushes between pops keep the order
MyQueue()
push(1)
push(2)
pop() -> 1
push(3)
push(4)
pop() -> 2
peek() -> 3
push(5)
pop() -> 3
pop() -> 4
pop() -> 5
empty() -> true

# emptied, then used again
MyQueue()
push(1)
pop() -> 1
empty() -> true
push(2)
peek() -> 2
empty() -> false
pop() -> 2

# equal values
MyQueue()
push(7)
push(7)
push(3)
pop() -> 7
pop() -> 7
peek() -> 3

# every value from 1 to 9 in, then out in the same order
MyQueue()
push(1)
push(2)
push(3)
push(4)
push(5)
push(6)
push(7)
push(8)
push(9)
pop() -> 1
pop() -> 2
pop() -> 3
pop() -> 4
pop() -> 5
pop() -> 6
pop() -> 7
pop() -> 8
pop() -> 9
empty() -> true
