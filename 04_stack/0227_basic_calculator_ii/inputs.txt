# Test cases for Basic Calculator II — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like "1+" * 149999 + "1".
# Leave `expected` out to just see what your solution returns.

# Example 1
s = "3+2*2"
expected = 7

# Example 2
s = " 3/2 "
expected = 1

# Example 3
s = " 3+5 / 2 "
expected = 5

# single number
s = "42"
expected = 42

# zero
s = "0"
expected = 0

# largest integer
s = "2147483647"
expected = 2147483647

# multiplication goes before addition and subtraction
s = "2+3*4-5"
expected = 9

# minus goes from left to right
s = "10-2-3"
expected = 5

# times and divide go from left to right
s = "8/2*3"
expected = 12

# division after a minus truncates toward zero
s = "1-3/2"
expected = 0

# spaces everywhere
s = "  3 *  4 - 2 / 1 "
expected = 10

# stress: 299999 characters, 150000 ones added
s = "1+" * 149999 + "1"
expected = 150000
