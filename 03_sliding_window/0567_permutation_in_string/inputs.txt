# Test cases for Permutation in String — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like "a" * 10000.
# Leave `expected` out to just see what your solution returns.

# Example 1
s1 = "ab"
s2 = "eidbaooo"
expected = true

# Example 2
s1 = "ab"
s2 = "eidboaoo"
expected = false

# s1 longer than s2
s1 = "abc"
s2 = "ab"
expected = false

# single letters, equal
s1 = "a"
s2 = "a"
expected = true

# single letter missing from s2
s1 = "x"
s2 = "abc"
expected = false

# whole s2 is a permutation of s1
s1 = "abc"
s2 = "bca"
expected = true

# repeated letter in s1 must be matched as many times
s1 = "aab"
s2 = "abbab"
expected = false

# letters present but not next to each other
s1 = "ab"
s2 = "acb"
expected = false

# match at the very start of s2
s1 = "abc"
s2 = "cbaxyz"
expected = true

# match at the very end of s2
s1 = "ab"
s2 = "ooooba"
expected = true

# stress: s1 is 5000 letters, the only match is the last 5000 letters of s2
s1 = "b" + "a" * 4999
s2 = "a" * 9999 + "b"
expected = true

# stress: same sizes, s2 has no b at all
s1 = "b" + "a" * 4999
s2 = "a" * 10000
expected = false
