# Test cases for Next Permutation — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(1, 101)).
# Leave `expected` out to just see what your solution returns.
# `expected` is what `nums` must hold after the call: the method changes it in place.

# Example 1
nums = [1, 2, 3]
expected = [1, 3, 2]

# Example 2
nums = [3, 2, 1]
expected = [1, 2, 3]

# Example 3
nums = [1, 1, 5]
expected = [1, 5, 1]

# single element
nums = [1]
expected = [1]

# two elements, ascending
nums = [1, 2]
expected = [2, 1]

# all equal: already the last arrangement
nums = [2, 2, 2]
expected = [2, 2, 2]

# swap in the smallest larger value, not the nearest one
nums = [1, 3, 2]
expected = [2, 1, 3]

# equal neighbours at the end
nums = [1, 5, 5]
expected = [5, 1, 5]

# longer tail: after the swap the tail must end up ascending
nums = [5, 4, 7, 5, 3, 2]
expected = [5, 5, 2, 3, 4, 7]

# change in the middle, the start stays
nums = [2, 2, 0, 4, 3, 1]
expected = [2, 2, 1, 0, 3, 4]

# max length: last arrangement goes back to sorted
nums = list(range(100, 0, -1))
expected = list(range(1, 101))

# max length: only the last two values swap
nums = list(range(100))
expected = list(range(98)) + [99, 98]
