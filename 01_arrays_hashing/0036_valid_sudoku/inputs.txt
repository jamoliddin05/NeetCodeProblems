# Leave `expected` out to just see what your solution returns.
# Boards are written as 9 row strings; [list(row) for row in [...]] turns them into the 9 x 9 grid
# of one-character strings that LeetCode passes in.

board = [list(row) for row in ["53..7....", "6..195...", ".98....6.", "8...6...3", "4..8.3..1", "7...2...6", ".6....28.", "...419..5", "....8..79"]]
