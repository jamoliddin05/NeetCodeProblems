# Test cases for Insert Delete GetRandom O(1) — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each line is one call on the same object; the first line creates it.
# `-> value` after a call is the value it must return; calls without `->` aren't checked.
# Values are JSON (true, false, null, "text") or Python expressions.
# getRandom() is checked only where the set holds one element, because only then the answer is fixed.

# Example 1
RandomizedSet()
insert(1) -> true
remove(2) -> false
insert(2) -> true
getRandom()
remove(1) -> true
insert(2) -> false
getRandom() -> 2

# remove from an empty set
RandomizedSet()
remove(1) -> false
insert(1) -> true
getRandom() -> 1

# value inserted twice, removed twice
RandomizedSet()
insert(3) -> true
insert(3) -> false
remove(3) -> true
remove(3) -> false
insert(3) -> true
getRandom() -> 3

# the value at the first position is still found
RandomizedSet()
insert(5) -> true
insert(5) -> false
insert(6) -> true
remove(5) -> true
remove(5) -> false
getRandom() -> 6

# remove the last inserted value
RandomizedSet()
insert(1) -> true
insert(2) -> true
remove(2) -> true
remove(2) -> false
getRandom() -> 1
insert(2) -> true
remove(1) -> true
getRandom() -> 2

# remove the first value, then the value that took its place
RandomizedSet()
insert(1) -> true
insert(2) -> true
insert(3) -> true
remove(1) -> true
remove(3) -> true
getRandom() -> 2
remove(2) -> true
remove(1) -> false
remove(3) -> false
insert(1) -> true
getRandom() -> 1

# getRandom does not remove anything
RandomizedSet()
insert(7) -> true
getRandom() -> 7
getRandom() -> 7
getRandom() -> 7
insert(7) -> false
remove(7) -> true

# many values, removed in mixed order down to one
RandomizedSet()
insert(10) -> true
insert(20) -> true
insert(30) -> true
insert(40) -> true
insert(50) -> true
getRandom()
remove(30) -> true
remove(10) -> true
remove(50) -> true
getRandom()
remove(20) -> true
getRandom() -> 40
insert(20) -> true
remove(40) -> true
getRandom() -> 20

# zero and the 32-bit extremes
RandomizedSet()
insert(0) -> true
insert(-2147483648) -> true
insert(2147483647) -> true
remove(0) -> true
remove(0) -> false
remove(-2147483648) -> true
getRandom() -> 2147483647
insert(0) -> true
remove(2147483647) -> true
getRandom() -> 0
