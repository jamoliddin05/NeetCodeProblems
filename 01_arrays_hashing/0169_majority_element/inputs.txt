# Test cases for Majority Element — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [7] * 25001.
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [3, 2, 3]
expected = 3

# Example 2
nums = [2, 2, 1, 1, 1, 2, 2]
expected = 2

# single element
nums = [5]
expected = 5

# two equal elements
nums = [7, 7]
expected = 7

# majority only by one: alternating values
nums = [6, 5, 6, 5, 6]
expected = 6

# majority comes last, after all the other values
nums = [1, 2, 3, 4, 5, 5, 5, 5, 5]
expected = 5

# zero is the majority
nums = [0, -1, 0]
expected = 0

# values at the constraint extremes
nums = [-1000000000, 1000000000, -1000000000]
expected = -1000000000

# stress: 50000 numbers, the majority fills the second half
nums = list(range(24999)) + [-1] * 25001
expected = -1
