# Test cases for Rotate Array — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(100000)).
# Leave `expected` out to just see what your solution returns.
# `expected` is what `nums` must hold after the call: the method changes it in place.

# Example 1
nums = [1, 2, 3, 4, 5, 6, 7]
k = 3
expected = [5, 6, 7, 1, 2, 3, 4]

# Example 2
nums = [-1, -100, 3, 99]
k = 2
expected = [3, 99, -1, -100]

# k = 0: nothing moves
nums = [1, 2, 3]
k = 0
expected = [1, 2, 3]

# k equal to the length: back where it started
nums = [1, 2, 3]
k = 3
expected = [1, 2, 3]

# k larger than the length
nums = [1, 2, 3]
k = 4
expected = [3, 1, 2]

# single element, largest k
nums = [1]
k = 100000
expected = [1]

# two elements
nums = [1, 2]
k = 1
expected = [2, 1]

# k one less than the length: same as one step to the left
nums = [1, 2, 3, 4, 5]
k = 4
expected = [2, 3, 4, 5, 1]

# k and the length share a factor
nums = [1, 2, 3, 4, 5, 6]
k = 2
expected = [5, 6, 1, 2, 3, 4]

# 32-bit extremes
nums = [-2147483648, 2147483647, 0]
k = 1
expected = [0, -2147483648, 2147483647]

# stress: 10^5 numbers, k = 10^5 - 1 (one step left)
nums = list(range(100000))
k = 99999
expected = list(range(1, 100000)) + [0]

# stress: 99999 numbers, k = 10^5 (one step right)
nums = list(range(99999))
k = 100000
expected = [99998] + list(range(99998))
