# Test cases for Subarray Sum Equals K — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [0] * 20000.
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [1, 1, 1]
k = 2
expected = 2

# Example 2
nums = [1, 2, 3]
k = 3
expected = 2

# single element equal to k
nums = [5]
k = 5
expected = 1

# single element not equal to k
nums = [5]
k = 3
expected = 0

# k larger than the whole sum: no subarray
nums = [1, 2, 3]
k = 7
expected = 0

# only the whole array works
nums = [1, 2, 3]
k = 6
expected = 1

# negatives: a growing window does not always grow the sum
nums = [1, -1, 0]
k = 0
expected = 3

# all zeros with k = 0: every subarray counts
nums = [0, 0, 0, 0]
k = 0
expected = 10

# negative k
nums = [-1, -1, 1]
k = -1
expected = 3

# several overlapping subarrays
nums = [3, 4, 7, 2, -3, 1, 4, 2]
k = 7
expected = 4

# stress: 20000 zeros, k = 0: every subarray counts
nums = [0] * 20000
k = 0
expected = 200010000
