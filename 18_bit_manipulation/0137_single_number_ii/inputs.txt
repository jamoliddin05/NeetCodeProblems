# Test cases for Single Number II — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(10)) * 3.
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [2,2,3,2]
expected = 3

# Example 2
nums = [0,1,0,1,0,1,99]
expected = 99

# single element
nums = [7]
expected = 7

# single value is negative
nums = [-2,-2,1,1,-4,1,4,-2,4,4]
expected = -4

# single value is -1: every bit set
nums = [5,5,5,-1]
expected = -1

# triples are negative, single is positive
nums = [-1,-1,-1,5]
expected = 5

# single value is zero
nums = [3,0,3,3]
expected = 0

# single value is -2^31: only the sign bit set
nums = [2147483647,-2147483648,2147483647,2147483647]
expected = -2147483648

# single value is 2^31 - 1, triples are -2^31
nums = [-2147483648,-2147483648,2147483647,-2147483648]
expected = 2147483647

# largest size: 9999 values three times each, single value in the middle
nums = list(range(-4999, 5000)) * 2 + [-987654321] + list(range(4999, -5000, -1))
expected = -987654321
