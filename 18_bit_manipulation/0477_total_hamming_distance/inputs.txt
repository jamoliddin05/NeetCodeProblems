# Test cases for Total Hamming Distance — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(10000)).
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [4,14,2]
expected = 6

# Example 2
nums = [4,14,4]
expected = 4

# single element: no pairs
nums = [5]
expected = 0

# all equal
nums = [7,7,7,7]
expected = 0

# zero and the largest value 10^9 (13 bits set)
nums = [0,1000000000]
expected = 13

# 0 to 3: every pair of two-bit numbers
nums = [0,1,2,3]
expected = 8

# powers of two: every pair differs in two bits
nums = [1,2,4,8]
expected = 12

# largest size: 5000 zeros and 5000 numbers with 29 bits set
nums = [0] * 5000 + [536870911] * 5000
expected = 725000000

# largest size: 0 to 9999
nums = list(range(10000))
expected = 338133248
