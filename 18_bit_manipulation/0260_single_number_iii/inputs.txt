# Test cases for Single Number III — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(10)) * 2 + [-1, 99].
# Leave `expected` out to just see what your solution returns.
# The two values may be returned in any order.

# Example 1
nums = [1,2,1,3,2,5]
expected = [3,5]

# Example 2
nums = [-1,0]
expected = [-1,0]

# Example 3
nums = [0,1]
expected = [1,0]

# both values negative
nums = [-5,7,-3,7]
expected = [-5,-3]

# a value and its negative
nums = [6,2,-6,2]
expected = [-6,6]

# values differ only in the lowest bit
nums = [12,6,12,7]
expected = [6,7]

# values differ only in bit 30
nums = [1,9,9,1073741825]
expected = [1,1073741825]

# 32-bit extremes
nums = [-2147483648,9,2147483647,9]
expected = [-2147483648,2147483647]

# -2^31 and 0
nums = [4,-2147483648,4,0]
expected = [-2147483648,0]

# largest size: 14999 pairs, the two values at the ends
nums = [-1000000] + list(range(1, 15000)) + list(range(14999, 0, -1)) + [1000000]
expected = [-1000000,1000000]
