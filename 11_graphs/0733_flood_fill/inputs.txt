# Test cases for Flood Fill — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [[0] * 50 for _ in range(50)].
# Leave `expected` out to just see what your solution returns.

# Example 1
image = [[1,1,1],[1,1,0],[1,0,1]]
sr = 1
sc = 1
color = 2
expected = [[2,2,2],[2,2,0],[2,0,1]]

# Example 2
image = [[0,0,0],[0,0,0]]
sr = 0
sc = 0
color = 0
expected = [[0,0,0],[0,0,0]]

# single pixel
image = [[0]]
sr = 0
sc = 0
color = 5
expected = [[5]]

# new color equals the start color: must still stop
image = [[1,1],[1,0]]
sr = 0
sc = 0
color = 1
expected = [[1,1],[1,0]]

# diagonal neighbours are not connected
image = [[1,0,1],[0,1,0],[1,0,1]]
sr = 1
sc = 1
color = 3
expected = [[1,0,1],[0,3,0],[1,0,1]]

# same color in a separate area stays unchanged
image = [[2,2,0,2],[2,0,0,2]]
sr = 0
sc = 0
color = 5
expected = [[5,5,0,2],[5,0,0,2]]

# area winds around a wall, start in a corner
image = [[1,1,1],[0,0,1],[1,1,1]]
sr = 2
sc = 0
color = 9
expected = [[9,9,9],[0,0,9],[9,9,9]]

# new color is already used by the neighbours
image = [[1,2],[2,1]]
sr = 0
sc = 0
color = 2
expected = [[2,2],[2,1]]

# start color 0, largest allowed new color
image = [[0,0],[0,7]]
sr = 1
sc = 0
color = 65535
expected = [[65535,65535],[65535,7]]

# single row, start at the right end
image = [[3,3,4,3,3]]
sr = 0
sc = 4
color = 1
expected = [[3,3,4,1,1]]

# max size 50 x 50, one color: every pixel changes
image = [[0] * 50 for _ in range(50)]
sr = 49
sc = 49
color = 1
expected = [[1] * 50 for _ in range(50)]
