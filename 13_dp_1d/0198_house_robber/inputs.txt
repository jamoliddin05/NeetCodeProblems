# Test cases for House Robber — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [400] * 100.
# `random` in an expression is seeded from the expression's text, so the data is the same on every run.
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [1, 2, 3, 1]
expected = 4

# Example 2
nums = [2, 7, 9, 3, 1]
expected = 12

# single house
nums = [5]
expected = 5

# two houses: only one can be robbed
nums = [3, 10]
expected = 10

# all houses empty
nums = [0, 0, 0, 0]
expected = 0

# skipping two houses in a row is best
nums = [2, 1, 1, 2]
expected = 4

# neither all even nor all odd positions
nums = [1, 3, 1, 3, 100]
expected = 103

# the richest house is not taken
nums = [5, 6, 5]
expected = 10

# 100 houses with 400 each: plain recursion is too slow
nums = [400] * 100
expected = 20000

# 100 random houses
nums = [random.randint(0, 400) for _ in range(100)]
expected = 11675
