# Test cases for Coin Change — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(2, 25, 2)).
# Leave `expected` out to just see what your solution returns.

# Example 1
coins = [1, 2, 5]
amount = 11
expected = 3

# Example 2
coins = [2]
amount = 3
expected = -1

# Example 3
coins = [1]
amount = 0
expected = 0

# largest coin first uses too many coins
coins = [1, 3, 4]
amount = 6
expected = 2

# largest coin first gets stuck
coins = [3, 7]
amount = 9
expected = 3

# coin at the 2^31 - 1 limit, larger than the amount
coins = [2147483647]
amount = 2
expected = -1

# coins not sorted
coins = [10, 1, 25, 5]
amount = 30
expected = 2

# 12 even coins, odd amount: no answer, plain recursion is too slow
coins = list(range(2, 25, 2))
amount = 9999
expected = -1

# largest amount: 400 coins of 25
coins = [5, 10, 25]
amount = 10000
expected = 400

# many ways to combine the coins: plain recursion is too slow
coins = [186, 419, 83, 408]
amount = 6249
expected = 20
