# Test cases for Longest Increasing Subsequence — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(-1250, 1250)).
# `random` in an expression is seeded from the expression's text, so the data is the same on every run.
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [10, 9, 2, 5, 3, 7, 101, 18]
expected = 4

# Example 2
nums = [0, 1, 0, 3, 2, 3]
expected = 4

# Example 3
nums = [7, 7, 7, 7, 7, 7, 7]
expected = 1

# single element
nums = [5]
expected = 1

# strictly decreasing
nums = [5, 4, 3, 2, 1]
expected = 1

# equal neighbours do not count as increasing
nums = [1, 2, 2, 3]
expected = 3

# negative numbers
nums = [-2, -1, -3, 0, -5, 1]
expected = 4

# starting from the first element is not best
nums = [3, 4, -1, 0, 1, 2]
expected = 4

# the best subsequence skips around
nums = [0, 8, 4, 12, 2, 10, 6, 14, 1, 9, 5, 13, 3, 11, 7, 15]
expected = 6

# 2500 increasing numbers: plain recursion is too slow
nums = list(range(-1250, 1250))
expected = 2500

# 2500 random numbers
nums = [random.randint(-10**4, 10**4) for _ in range(2500)]
expected = 93
