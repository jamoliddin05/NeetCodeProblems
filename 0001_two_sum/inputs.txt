# Test cases for Two Sum — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(1000)).
# Leave `expected` out to just see what your solution returns.
# The answer must list the smaller index first, and every input has exactly one answer.

# Example 1
nums = [3, 4, 5, 6]
target = 7
expected = [0, 1]

# Example 2
nums = [4, 5, 6]
target = 10
expected = [0, 2]

# Example 3
nums = [5, 5]
target = 10
expected = [0, 1]

# half of target appears once, can't pair with itself
nums = [3, 2, 4]
target = 6
expected = [1, 2]

# negatives summing to zero
nums = [-3, 4, 3, 90]
target = 0
expected = [0, 2]

# two zeros, far apart
nums = [0, 4, 3, 0]
target = 0
expected = [0, 3]

# all negative, negative target
nums = [-1, -2, -3, -4, -5]
target = -8
expected = [2, 4]

# pair is the last two elements
nums = [1, 2, 3, 4, 5, 6]
target = 11
expected = [4, 5]

# larger value comes first in the array
nums = [7, 1, 2]
target = 3
expected = [1, 2]

# values at the constraint extremes
nums = [10000000, 0, -10000000]
target = 10000000
expected = [0, 1]

# max size: 1000 elements, pair at the very end
nums = list(range(1000))
target = 1997
expected = [998, 999]
