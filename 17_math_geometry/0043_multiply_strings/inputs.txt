# Test cases for Multiply Strings — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like "9" * 200.
# Leave `expected` out to just see what your solution returns.

# Example 1
num1 = "2"
num2 = "3"
expected = "6"

# Example 2
num1 = "123"
num2 = "456"
expected = "56088"

# zero times a long number: no leading zeros
num1 = "0"
num2 = "12345"
expected = "0"

# long number times zero
num1 = "9133"
num2 = "0"
expected = "0"

# both zero
num1 = "0"
num2 = "0"
expected = "0"

# one digit each, with a carry
num1 = "9"
num2 = "9"
expected = "81"

# carries through every digit
num1 = "999"
num2 = "999"
expected = "998001"

# times one
num1 = "1"
num2 = "123456789"
expected = "123456789"

# zeros inside and at the end of the numbers
num1 = "1020"
num2 = "300"
expected = "306000"

# 20 digits each
num1 = "12345678901234567890"
num2 = "98765432109876543210"
expected = "1219326311370217952237463801111263526900"

# different lengths: 200 digits times 1 digit
num1 = "9" * 200
num2 = "9"
expected = "8" + "9" * 199 + "1"

# largest size: 200 nines times 200 nines
num1 = "9" * 200
num2 = "9" * 200
expected = "9" * 199 + "8" + "0" * 199 + "1"
