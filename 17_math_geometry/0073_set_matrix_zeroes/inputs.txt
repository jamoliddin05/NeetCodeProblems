# Test cases for Set Matrix Zeroes — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [[1] * 200 for _ in range(200)].
# Leave `expected` out to just see what your solution returns.

# Example 1
matrix = [[1,1,1],[1,0,1],[1,1,1]]
expected = [[1,0,1],[0,0,0],[1,0,1]]

# Example 2
matrix = [[0,1,2,0],[3,4,5,2],[1,3,1,5]]
expected = [[0,0,0,0],[0,4,5,0],[0,3,1,0]]

# 1 x 1 zero
matrix = [[0]]
expected = [[0]]

# 1 x 1 non-zero
matrix = [[5]]
expected = [[5]]

# no zeros: nothing changes
matrix = [[1,2],[3,4]]
expected = [[1,2],[3,4]]

# zero in the top-left corner
matrix = [[0,1,2],[3,4,5],[6,7,8]]
expected = [[0,0,0],[0,4,5],[0,7,8]]

# zero in the first row only: column 0 must stay
matrix = [[1,0,3],[4,5,6]]
expected = [[0,0,0],[4,0,6]]

# zero in the first column only: row 0 must stay
matrix = [[1,2,3],[0,5,6],[7,8,9]]
expected = [[0,2,3],[0,0,0],[0,8,9]]

# single row
matrix = [[1,0,2,3]]
expected = [[0,0,0,0]]

# single column
matrix = [[1],[0],[2]]
expected = [[0],[0],[0]]

# 32-bit extremes next to a zero
matrix = [[-2147483648,2147483647,-1],[0,1,2147483647]]
expected = [[0,2147483647,-1],[0,0,0]]

# largest size: 200 x 200 with two zeros
matrix = [[0 if (i, j) in ((5, 7), (150, 199)) else i * 200 + j + 1 for j in range(200)] for i in range(200)]
expected = [[0 if i in (5, 150) or j in (7, 199) else i * 200 + j + 1 for j in range(200)] for i in range(200)]
