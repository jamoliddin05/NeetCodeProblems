piles = [3,6,7,11]

# one pile of 10^9, one hour

# one pile, hours don't divide it evenly
piles = [10]

# speed divides every pile evenly

# plenty of time: the answer is 1, not 0
piles = [3, 6, 7, 11]

expected = 1

# one hour less than speed 1 needs
piles = [3, 6, 7, 11]

expected = 2

# stress: 9999 small piles take an hour each, leaving 2 hours for the 10^9 pile

# stress: 10^4 piles of 10^9 and h = 10^9, so 10^5 hours per pile
