# Leave `expected` out to just see what your solution returns.

nums = [-1,0,3,5,9,12]

# single element, missing
nums = [5]

expected = -1

# two elements, target is the second
nums = [1, 3]

# target is the first element

expected = 0

# target smaller than every element
nums = [2, 4, 6, 8]

expected = -1

# target larger than every element

expected = -1

# all negative
nums = [-10, -8, -6, -4, -2]

expected = 3

# max size: target is the last element

target = 9999
