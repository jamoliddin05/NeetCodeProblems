# Test cases for Container With Most Water — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [10000] * 100000.
# Leave `expected` out to just see what your solution returns.

# Example 1
heights = [1, 7, 2, 5, 4, 7, 3, 6]
expected = 36

# Example 2
heights = [2, 2, 2]
expected = 4

# smallest input: two bars
heights = [1, 1]
expected = 1

# zero-height bar holds nothing
heights = [0, 5]
expected = 0

# all zeros
heights = [0, 0, 0, 0]
expected = 0

# tallest bar is not part of the answer
heights = [1, 2, 1000, 1, 2]
expected = 6

# two tall bars side by side lose to a wide short pair
heights = [3, 1, 1, 9, 9, 1, 1, 3]
expected = 21

# increasing heights
heights = [1, 2, 3, 4, 5]
expected = 6

# decreasing heights
heights = [5, 4, 3, 2, 1]
expected = 6

# equal outer bars with lower ones between
heights = [4, 1, 1, 1, 4]
expected = 16

# classic mixed case
heights = [1, 8, 6, 2, 5, 4, 8, 3, 7]
expected = 49

# max height, two bars
heights = [10000, 10000]
expected = 10000

# stress: 10^5 bars of max height, the two ends win
heights = [10000] * 100000
expected = 999990000

# stress: 10^5 bars, only the two ends are non-zero
heights = [5] + [0] * 99998 + [5]
expected = 499995
