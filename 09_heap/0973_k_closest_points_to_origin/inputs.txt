# Test cases for K Closest Points to Origin — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [[i, 0] for i in range(10)].
# Leave `expected` out to just see what your solution returns.
# The points may come back in any order.

# Example 1
points = [[1,3],[-2,2]]
k = 1
expected = [[-2,2]]

# Example 2
points = [[3,3],[5,-1],[-2,4]]
k = 2
expected = [[3,3],[-2,4]]

# single point at the coordinate extremes
points = [[10000, -10000]]
k = 1
expected = [[10000, -10000]]

# k = n: every point
points = [[1, 2], [-3, 4], [0, -5]]
k = 3
expected = [[1, 2], [-3, 4], [0, -5]]

# the origin itself
points = [[5, 5], [0, 0], [1, 1]]
k = 1
expected = [[0, 0]]

# negative coordinates
points = [[-1, -1], [2, 2], [-3, 0]]
k = 1
expected = [[-1, -1]]

# sum of absolute values gives the wrong answer
points = [[3, 0], [2, 2]]
k = 1
expected = [[2, 2]]

# equal distances inside the answer
points = [[1, 0], [0, 1], [-1, 0], [3, 3]]
k = 3
expected = [[1, 0], [0, 1], [-1, 0]]

# the same point twice
points = [[2, 2], [2, 2], [1, 1], [3, 3]]
k = 3
expected = [[1, 1], [2, 2], [2, 2]]

# stress: 10^4 points, the closest half wanted
points = [[i, -i] for i in reversed(range(10000))]
k = 5000
expected = [[i, -i] for i in range(5000)]
