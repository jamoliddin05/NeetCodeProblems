# Test cases for Last Stone Weight — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [1000] * 30.
# Leave `expected` out to just see what your solution returns.

# Example 1
stones = [2,7,4,1,8,1]
expected = 1

# Example 2
stones = [1]
expected = 1

# two equal stones: nothing is left
stones = [5, 5]
expected = 0

# two different stones
stones = [3, 10]
expected = 7

# every stone gets destroyed
stones = [2, 3, 2, 3]
expected = 0

# odd number of equal stones: one survives
stones = [4, 4, 4]
expected = 4

# leftover stays the heaviest for several turns
stones = [1, 1, 2, 10]
expected = 6

# leftover becomes the lightest stone
stones = [9, 8, 7, 1]
expected = 5

# max length: 30 equal stones
stones = [6] * 30
expected = 0

# max length and max weight: 29 stones of 1000 and one of 1
stones = [1000] * 29 + [1]
expected = 999
