# Values are JSON (true, false, null, "text") or Python expressions like ["A"] * 10000.
# Leave `expected` out to just see what your solution returns.

# Example 3
tasks = ["A","A","A", "B","B","B"]
n = 3
expected = 10

# n = 0: no waiting at all
tasks = ["A","A","A","B","B","B"]
n = 0
expected = 6

# single task, long cooldown
tasks = ["A"]
n = 100
expected = 1

# one label only: idle after every task but the last
tasks = ["A","A","A"]
n = 2
expected = 7

# enough other labels to fill every gap: no idle
tasks = ["A","A","A","B","C","D","E","F","G"]
n = 2
expected = 9

# several labels tied for the most frequent
tasks = ["A","A","B","B","C","C","D"]
n = 4
expected = 8

# 26 different labels, max cooldown
tasks = list("ABCDEFGHIJKLMNOPQRSTUVWXYZ")
n = 100
expected = 26

# 26 labels, 384 of each, n = 25: gaps fit exactly
tasks = list("ABCDEFGHIJKLMNOPQRSTUVWXYZ") * 384
n = 25
expected = 9984

# stress: 10^4 copies of one label, n = 100
tasks = ["A"] * 10000
n = 100
