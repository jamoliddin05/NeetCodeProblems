# Test cases for Partition Labels — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like "ab" * 250.
# Leave `expected` out to just see what your solution returns.

# Example 1
s = "ababcbacadefegdehijhklij"
expected = [9, 7, 8]

# Example 2
s = "eccbbbbdec"
expected = [10]

# the statement's "ababcc"
s = "ababcc"
expected = [4, 2]

# single letter
s = "a"
expected = [1]

# all letters different: every letter is its own part
s = "abc"
expected = [1, 1, 1]

# one letter repeated: one part
s = "aaaa"
expected = [4]

# first letter comes back at the end
s = "abcda"
expected = [5]

# a letter inside the part pushes its end further
s = "abacbdd"
expected = [5, 2]

# max length: 25 blocks of 20 equal letters
s = "".join(chr(97 + i // 20) for i in range(500))
expected = [20] * 25

# max length: the alphabet repeated, one part
s = "".join(chr(97 + i % 26) for i in range(500))
expected = [500]
