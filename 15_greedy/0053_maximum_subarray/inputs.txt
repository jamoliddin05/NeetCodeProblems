# Test cases for Maximum Subarray — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [-1] * 1000.
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [-2, 1, -3, 4, -1, 2, 1, -5, 4]
expected = 6

# Example 2
nums = [1]
expected = 1

# Example 3
nums = [5, 4, -1, 7, 8]
expected = 23

# single negative element
nums = [-5]
expected = -5

# all negative: the answer is the largest element, not 0
nums = [-3, -1, -2]
expected = -1

# negatives and a zero
nums = [-2, 0, -1]
expected = 0

# best part sits in the middle
nums = [-1, 3, -1, 2, -5]
expected = 4

# a big dip that is still worth crossing
nums = [8, -19, 5, -4, 20]
expected = 21

# value extremes: two of -10^4
nums = [-10000, -10000]
expected = -10000

# stress: 10^5 elements, all -10^4 except a -1 at the end
nums = [-10000] * 99999 + [-1]
expected = -1

# stress: 10^5 elements, all 10^4: the whole array wins
nums = [10000] * 100000
expected = 1000000000
