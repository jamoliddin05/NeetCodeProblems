# Test cases for Gas Station — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [1] * 1000.
# Leave `expected` out to just see what your solution returns.

# Example 1
gas = [1, 2, 3, 4, 5]
cost = [3, 4, 5, 1, 2]
expected = 3

# Example 2
gas = [2, 3, 4]
cost = [3, 4, 3]
expected = -1

# single station, gas equals cost: the tank may reach exactly 0
gas = [3]
cost = [3]
expected = 0

# single station, not enough gas
gas = [1]
cost = [2]
expected = -1

# the answer is station 0
gas = [3, 1, 1]
cost = [1, 2, 2]
expected = 0

# the answer is the last station
gas = [1, 1, 3]
cost = [2, 2, 1]
expected = 2

# spare gas at the first stations still runs out later
gas = [5, 1, 2, 3, 4]
cost = [4, 4, 1, 5, 1]
expected = 4

# no answer: total gas is 1 less than total cost
gas = [4, 1, 1]
cost = [1, 2, 4]
expected = -1

# total gas exactly equals total cost
gas = [2, 3, 1]
cost = [3, 1, 2]
expected = 1

# stress: 10^5 stations, no answer, every start fails only at the last station
gas = [1] * 100000
cost = [1] * 99999 + [2]
expected = -1

# stress: 10^5 stations, only the last station works
gas = [1] * 99998 + [0, 2]
cost = [1] * 100000
expected = 99999
