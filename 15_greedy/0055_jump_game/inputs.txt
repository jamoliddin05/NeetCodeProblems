# Test cases for Jump Game — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [1] * 1000.
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [2, 3, 1, 1, 4]
expected = true

# Example 2
nums = [3, 2, 1, 0, 4]
expected = false

# single element: already on the last index
nums = [0]
expected = true

# stuck on the first index
nums = [0, 1]
expected = false

# a zero on the last index doesn't matter
nums = [1, 1, 1, 0]
expected = true

# zeros that a longer jump goes over
nums = [3, 0, 0, 0]
expected = true

# a zero that every path lands on
nums = [1, 1, 0, 1]
expected = false

# first jump is longer than the whole array
nums = [100000, 0, 0, 0]
expected = true

# stress: 10^4 elements, every jump ends on the same zero
nums = list(range(9998, 0, -1)) + [0, 0]
expected = false

# stress: 10^4 elements, a zero at every other index, each one jumped over
nums = [2, 0] * 4999 + [1, 0]
expected = true
