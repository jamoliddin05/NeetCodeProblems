# Test cases for Lemonade Change — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [5, 10] * 500.
# Leave `expected` out to just see what your solution returns.

# Example 1
bills = [5, 5, 5, 10, 20]
expected = true

# Example 2
bills = [5, 5, 10, 10, 20]
expected = false

# single $5 bill: no change needed
bills = [5]
expected = true

# first customer pays $10: nothing to give back
bills = [10]
expected = false

# one $5 is not enough for a $20
bills = [5, 20]
expected = false

# $20 paid back with three $5 bills
bills = [5, 5, 5, 20]
expected = true

# a $10 can't be given back for a $10
bills = [5, 10, 10]
expected = false

# a $20 you hold can't be given back as change
bills = [5, 5, 5, 20, 20]
expected = false

# $20 with $10 + $5 keeps $5 bills for the $10 customers after it
bills = [5, 5, 5, 5, 10, 20, 10, 10]
expected = true

# stress: 10^5 customers, every $20 gets $10 + $5 back
bills = [5, 5, 10, 20] * 25000
expected = true

# stress: 10^5 customers, many $10 bills but no $5 left for the last $20
bills = [5, 10] * 49999 + [20, 5]
expected = false
