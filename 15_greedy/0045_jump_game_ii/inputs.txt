# Test cases for Jump Game II — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [1] * 1000.
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [2, 3, 1, 1, 4]
expected = 2

# Example 2
nums = [2, 3, 0, 1, 4]
expected = 2

# single element: no jump needed
nums = [0]
expected = 0

# first jump can reach the last index
nums = [10, 1, 1, 1]
expected = 1

# all ones: one step per jump
nums = [1, 1, 1, 1, 1]
expected = 4

# landing as far as possible is not the best first jump
nums = [4, 1, 1, 3, 1, 1, 1]
expected = 2

# zeros that must be jumped over
nums = [2, 0, 2, 0, 1]
expected = 2

# stress: 10^4 ones, 9999 jumps
nums = [1] * 10000
expected = 9999

# stress: 10^4 elements of 1000, every jump covers 1000 indices
nums = [1000] * 10000
expected = 10
