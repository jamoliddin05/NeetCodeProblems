# Test cases for Copy List with Random Pointer — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# A list is written as [[val, random_index], ...]; random_index is null when random points nowhere.
# tests.py also fails a copy that reuses original nodes or leaves the original list changed.
# Values are JSON (true, false, null, "text") or Python expressions like [[i, None] for i in range(10)].
# Leave `expected` out to just see what your solution returns.

# Example 1
head = [[7,null],[13,0],[11,4],[10,2],[1,0]]
expected = [[7,null],[13,0],[11,4],[10,2],[1,0]]

# Example 2
head = [[1,1],[2,1]]
expected = [[1,1],[2,1]]

# Example 3: equal values, so nodes can't be told apart by value
head = [[3,null],[3,0],[3,null]]
expected = [[3,null],[3,0],[3,null]]

# empty list
head = []
expected = []

# single node, random is null
head = [[-10000,null]]
expected = [[-10000,null]]

# single node, random points to itself
head = [[10000,0]]
expected = [[10000,0]]

# no random pointers at all
head = [[1,null],[2,null],[3,null]]
expected = [[1,null],[2,null],[3,null]]

# every random points forward to the last node
head = [[1,3],[2,3],[3,3],[4,3]]
expected = [[1,3],[2,3],[3,3],[4,3]]

# max size: 1000 nodes, each random points to the mirror position
head = [[i, 999 - i] for i in range(1000)]
expected = [[i, 999 - i] for i in range(1000)]
