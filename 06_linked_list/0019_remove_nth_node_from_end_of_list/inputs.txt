# Test cases for Remove Nth Node From End of List — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# A linked list is written as the list of its values, head first: [1, 2, 3]; [] is an empty list.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(30)).
# Leave `expected` out to just see what your solution returns.

# Example 1
head = [1,2,3,4,5]
n = 2
expected = [1,2,3,5]

# Example 2: the only node is removed
head = [1]
n = 1
expected = []

# Example 3
head = [1,2]
n = 1
expected = [1]

# two nodes, remove the head
head = [1, 2]
n = 2
expected = [2]

# remove the head: n equals the length
head = [1, 2, 3]
n = 3
expected = [2, 3]

# remove the last node
head = [10, 20, 30, 40]
n = 1
expected = [10, 20, 30]

# repeated values: remove by position, not by value
head = [1, 2, 1, 2]
n = 2
expected = [1, 2, 2]

# max length 30, remove the middle node
head = list(range(30))
n = 15
expected = list(range(15)) + list(range(16, 30))

# max length 30, remove the head
head = list(range(1, 31))
n = 30
expected = list(range(2, 31))
