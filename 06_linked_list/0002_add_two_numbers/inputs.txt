# Test cases for Add Two Numbers — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# A linked list is written as the list of its values, head first: [2, 4, 3] is the number 342.
# Values are JSON (true, false, null, "text") or Python expressions like [9] * 100.
# Leave `expected` out to just see what your solution returns.

# Example 1
l1 = [2,4,3]
l2 = [5,6,4]
expected = [7,0,8]

# Example 2
l1 = [0]
l2 = [0]
expected = [0]

# Example 3
l1 = [9,9,9,9,9,9,9]
l2 = [9,9,9,9]
expected = [8,9,9,9,0,0,0,1]

# carry makes the result one digit longer
l1 = [5]
l2 = [5]
expected = [0, 1]

# first list shorter, carry goes through the longer one
l1 = [1]
l2 = [9, 9, 9]
expected = [0, 0, 0, 1]

# second list shorter, no carry
l1 = [1, 2, 3]
l2 = [4]
expected = [5, 2, 3]

# one number is zero
l1 = [0]
l2 = [3, 2, 1]
expected = [3, 2, 1]

# max length: carry runs through 100 digits
l1 = [9] * 100
l2 = [1]
expected = [0] * 100 + [1]

# max length: both 100 digits, no carry
l1 = [1] * 100
l2 = [2] * 100
expected = [3] * 100
