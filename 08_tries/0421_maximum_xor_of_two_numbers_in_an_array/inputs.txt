# Test cases for Maximum XOR of Two Numbers in an Array — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(200000)).
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [3,10,5,25,2,8]
expected = 28

# Example 2
nums = [14,70,53,83,49,91,36,80,92,51,66,70]
expected = 127

# single number: i == j is allowed, so the answer is 0
nums = [7]
expected = 0

# all numbers equal
nums = [5,5,5]
expected = 0

# only zeros
nums = [0,0]
expected = 0

# zero and the largest allowed value
nums = [0,2147483647]
expected = 2147483647

# the largest number is not in the best pair
nums = [15,8,7]
expected = 15

# powers of two
nums = [1,2,4,8,16]
expected = 24

# the top bit comes from one number, all lower bits from the other
nums = [1073741824,1073741823]
expected = 2147483647

# max size: 0 .. 199999 holds 2^17 and 2^17 - 1, whose XOR is 2^18 - 1
nums = list(range(200000))
expected = 262143

# max size: one 0 among 199999 copies of 2^31 - 1
nums = [2147483647] * 199999 + [0]
expected = 2147483647
