# Test cases for Longest Common Prefix — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like ["a" * 200] * 200.
# Leave `expected` out to just see what your solution returns.

# Example 1
strs = ["flower","flow","flight"]
expected = "fl"

# Example 2
strs = ["dog","racecar","car"]
expected = ""

# single string: the answer is the whole string
strs = ["alone"]
expected = "alone"

# single empty string
strs = [""]
expected = ""

# one empty string among others
strs = ["abc","","ab"]
expected = ""

# all strings the same
strs = ["same","same","same"]
expected = "same"

# shortest string is the whole prefix, and it is not first
strs = ["interview","inter","internet"]
expected = "inter"

# first two agree, the last one differs at the first letter
strs = ["flower","flow","alpha"]
expected = ""

# two one-letter strings that differ
strs = ["a","b"]
expected = ""

# max size: 200 equal strings of length 200
strs = ["a" * 200] * 200
expected = "a" * 200

# max size: the last string differs only at its last letter
strs = ["a" * 200] * 199 + ["a" * 199 + "b"]
expected = "a" * 199
