# Test cases for Contains Duplicate — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(10**5)).
# Leave `expected` out to just see what your solution returns.

# Example 1
nums = [1, 2, 3, 3]
expected = true

# Example 2
nums = [1, 2, 3, 4]
expected = false

# empty array
nums = []
expected = false

# single element
nums = [7]
expected = false

# just two equal values
nums = [5, 5]
expected = true

# duplicate at the very first and last positions
nums = [1, 2, 3, 4, 5, 1]
expected = true

# every value the same
nums = [0, 0, 0, 0]
expected = true

# negatives with a duplicate
nums = [-1, -2, -3, -1]
expected = true

# same magnitude, opposite sign is not a duplicate
nums = [-3, 3, -2, 2, 0]
expected = false

# constraint extremes, all distinct
nums = [-1000000000, 1000000000, 0]
expected = false

# constraint extremes, repeated
nums = [1000000000, -1000000000, 1000000000]
expected = true

# stress: 10^5 distinct values
nums = list(range(100000))
expected = false

# stress: 10^5 values, the only duplicate is the last pair
nums = list(range(99999)) + [99998]
expected = true
