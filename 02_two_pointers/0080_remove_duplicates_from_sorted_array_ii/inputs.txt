# Test cases for Remove Duplicates from Sorted Array II — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [5] * 10.
# Leave `expected` out to just see what your solution returns.
# `expected` is the first k values of `nums` after the call, where k is the returned number;
# whatever is left after them is ignored, as on the site.

# Example 1
nums = [1, 1, 1, 2, 2, 3]
expected = [1, 1, 2, 2, 3]

# Example 2
nums = [0, 0, 1, 1, 1, 1, 2, 3, 3]
expected = [0, 0, 1, 1, 2, 3, 3]

# single element
nums = [1]
expected = [1]

# two equal values: both stay
nums = [1, 1]
expected = [1, 1]

# three equal values: one goes
nums = [1, 1, 1]
expected = [1, 1]

# no duplicates
nums = [1, 2, 3]
expected = [1, 2, 3]

# all equal, long run
nums = [5] * 10
expected = [5, 5]

# every value three times
nums = [1, 1, 1, 2, 2, 2, 3, 3, 3]
expected = [1, 1, 2, 2, 3, 3]

# negatives and zero
nums = [-3, -3, -3, -1, -1, 0, 0, 0]
expected = [-3, -3, -1, -1, 0, 0]

# values at the constraint extremes
nums = [-10000, -10000, -10000, 10000, 10000, 10000]
expected = [-10000, -10000, 10000, 10000]

# stress: 30000 numbers, each of 10000 values three times
nums = sorted(list(range(-5000, 5000)) * 3)
expected = sorted(list(range(-5000, 5000)) * 2)
