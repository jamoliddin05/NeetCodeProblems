# Test cases for Valid Palindrome II — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like "a" * 50000.
# Leave `expected` out to just see what your solution returns.

# Example 1
s = "aba"
expected = true

# Example 2
s = "abca"
expected = true

# Example 3
s = "abc"
expected = false

# single character
s = "a"
expected = true

# two different characters: delete either one
s = "ab"
expected = true

# already a palindrome, even length
s = "abba"
expected = true

# only deleting from the left side works
s = "ccbbc"
expected = true

# only deleting from the right side works
s = "cbbcc"
expected = true

# needs two deletions
s = "abcda"
expected = false

# first mismatch: both deletions look fine, only one works
s = "ebcbbececabbacecbbcbe"
expected = true

# stress: 10^5 characters, one extra letter next to the middle
s = "a" * 49999 + "bc" + "a" * 49999
expected = true

# stress: 99999 characters, "bcd" in the middle
s = "a" * 49998 + "bcd" + "a" * 49998
expected = false
