# Test cases for 4Sum — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like list(range(200)).
# Leave `expected` out to just see what your solution returns.
# Quadruplets may come in any order, and so may the numbers inside each quadruplet.

# Example 1
nums = [1, 0, -1, 0, -2, 2]
target = 0
expected = [[-2, -1, 1, 2], [-2, 0, 0, 2], [-1, 0, 0, 1]]

# Example 2
nums = [2, 2, 2, 2, 2]
target = 8
expected = [[2, 2, 2, 2]]

# single number: no quadruplet possible
nums = [0]
target = 0
expected = []

# exactly four numbers that add up
nums = [1, 2, 3, 4]
target = 10
expected = [[1, 2, 3, 4]]

# no quadruplet adds up
nums = [1, 2, 3, 4, 5]
target = 100
expected = []

# every position has duplicates: one quadruplet only
nums = [-1, -1, -1, -1, 1, 1, 1, 1]
target = 0
expected = [[-1, -1, 1, 1]]

# negative target: a number above the target can still be used
nums = [-1, -1, -1, -1]
target = -4
expected = [[-1, -1, -1, -1]]

# many quadruplets with shared numbers
nums = [-3, -2, -1, 0, 0, 1, 2, 3]
target = 0
expected = [[-3, -2, 2, 3], [-3, -1, 1, 3], [-3, 0, 0, 3], [-3, 0, 1, 2], [-2, -1, 0, 3], [-2, -1, 1, 2], [-2, 0, 0, 2], [-1, 0, 0, 1]]

# sum of four is beyond the 32-bit range
nums = [1000000000, 1000000000, 1000000000, 1000000000]
target = -294967296
expected = []

# stress: 200 distinct numbers, only 0 + 1 + 2 + 3 reaches 6
nums = list(range(200))
target = 6
expected = [[0, 1, 2, 3]]

# stress: 200 zeros, the one quadruplet must appear once
nums = [0] * 200
target = 0
expected = [[0, 0, 0, 0]]
