# Test cases for 3Sum — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [0] * 3000.
# Leave `expected` out to just see what your solution returns.
# Triplets may come in any order, and so may the numbers inside each triplet.

# Example 1
nums = [-1, 0, 1, 2, -1, -4]
expected = [[-1, -1, 2], [-1, 0, 1]]

# Example 2
nums = [0, 1, 1]
expected = []

# Example 3
nums = [0, 0, 0]
expected = [[0, 0, 0]]

# many zeros: still only one [0, 0, 0]
nums = [0, 0, 0, 0, 0]
expected = [[0, 0, 0]]

# all positive: no triplet can reach zero
nums = [1, 2, 3, 4, 5]
expected = []

# all negative
nums = [-5, -4, -3, -2, -1]
expected = []

# smallest array, unsorted
nums = [3, -1, -2]
expected = [[-2, -1, 3]]

# repeated values everywhere: each triplet only once
nums = [-2, 0, 0, 2, 2, -2, 0]
expected = [[-2, 0, 2], [0, 0, 0]]

# two 2s may share a triplet, but the single 3 can't be used twice
nums = [-4, 2, 2, 1, 3, -6]
expected = [[-4, 1, 3], [-4, 2, 2]]

# values at the constraint extremes
nums = [-100000, 100000, 0, 50000, 50000, -100000]
expected = [[-100000, 0, 100000], [-100000, 50000, 50000]]

# max length: 3000 zeros
nums = [0] * 3000
expected = [[0, 0, 0]]

# max length: only the two largest positives pair with the one negative
nums = list(range(1, 3000)) + [-5997]
expected = [[-5997, 2998, 2999]]
