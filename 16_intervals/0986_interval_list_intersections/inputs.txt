# Test cases for Interval List Intersections — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [[2 * i, 2 * i + 1] for i in range(100)].
# Leave `expected` out to just see what your solution returns.

# Example 1
firstList = [[0, 2], [5, 10], [13, 23], [24, 25]]
secondList = [[1, 5], [8, 12], [15, 24], [25, 26]]
expected = [[1, 2], [5, 5], [8, 10], [15, 23], [24, 24], [25, 25]]

# Example 2
firstList = [[1, 3], [5, 9]]
secondList = []
expected = []

# first list empty
firstList = []
secondList = [[1, 3]]
expected = []

# identical lists
firstList = [[1, 3], [5, 7]]
secondList = [[1, 3], [5, 7]]
expected = [[1, 3], [5, 7]]

# one interval covers several in the other list
firstList = [[0, 10]]
secondList = [[1, 2], [4, 5], [8, 12]]
expected = [[1, 2], [4, 5], [8, 10]]

# intervals that only touch give a single point
firstList = [[1, 3]]
secondList = [[3, 5]]
expected = [[3, 3]]

# no overlap at all
firstList = [[1, 2], [5, 6]]
secondList = [[3, 4], [7, 8]]
expected = []

# one interval meets two in the other list
firstList = [[1, 4], [6, 9]]
secondList = [[2, 7]]
expected = [[2, 4], [6, 7]]

# coordinates up to 10^9
firstList = [[0, 1000000000]]
secondList = [[999999999, 1000000000]]
expected = [[999999999, 1000000000]]

# max length: 1000 intervals each, every intersection is a single point
firstList = [[2 * i, 2 * i + 1] for i in range(1000)]
secondList = [[2 * i + 1, 2 * i + 2] for i in range(1000)]
expected = [[k, k] for k in range(1, 2000)]
