# Test cases for Merge Intervals — run them with: python tests.py
#
# One case per block, blocks separated by a blank line; the "# ..." line on top names the case.
# Each `name = value` line is an argument, `expected` is the correct answer.
# Values are JSON (true, false, null, "text") or Python expressions like [[i, i] for i in range(100)].
# Leave `expected` out to just see what your solution returns.
# The merged intervals may come in any order.

# Example 1
intervals = [[1, 3], [2, 6], [8, 10], [15, 18]]
expected = [[1, 6], [8, 10], [15, 18]]

# Example 2
intervals = [[1, 4], [4, 5]]
expected = [[1, 5]]

# Example 3
intervals = [[4, 7], [1, 4]]
expected = [[1, 7]]

# single interval
intervals = [[1, 3]]
expected = [[1, 3]]

# a gap of 1 is not an overlap
intervals = [[1, 2], [3, 4]]
expected = [[1, 2], [3, 4]]

# an interval inside the merged one must not shrink its end
intervals = [[1, 10], [2, 3], [4, 12]]
expected = [[1, 12]]

# one interval covers all the others
intervals = [[2, 3], [1, 10], [4, 5]]
expected = [[1, 10]]

# the same interval three times
intervals = [[2, 5], [2, 5], [2, 5]]
expected = [[2, 5]]

# zero-length intervals
intervals = [[5, 5], [0, 0], [1, 1], [1, 3]]
expected = [[0, 0], [1, 3], [5, 5]]

# unsorted, nothing overlaps
intervals = [[7, 8], [4, 5], [1, 2]]
expected = [[1, 2], [4, 5], [7, 8]]

# stress: 10^4 intervals in reverse order, each touches the next
intervals = [[i, i + 1] for i in range(9999, -1, -1)]
expected = [[0, 10000]]

# stress: 10^4 zero-length intervals in reverse order, none overlap
intervals = [[i, i] for i in range(9999, -1, -1)]
expected = [[i, i] for i in range(10000)]
